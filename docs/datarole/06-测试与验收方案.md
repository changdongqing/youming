### 2.1 DataScopeProcessor 测试

```java

import com.pig4cloud.pig.common.core.context.UserContext;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

import java.util.List;
import java.util.Set;

import static org.assertj.core.api.Assertions.*;
import static org.mockito.ArgumentMatchers.any;

/**
 * DataScopeProcessor 单元测试。
 *
 * <p>覆盖：多角色取最宽范围规则、短路逻辑、边界场景。

@DisplayName("数据权限处理器测试")
class DataScopeProcessorTest {

    private UserContextHolder userContextHolder;
    private DataScopeHandler dataScopeHandler;
    private DataScopeProcessor processor;

    @BeforeEach
    void setUp() {
        userContextHolder = Mockito.mock(UserContextHolder.class);
        dataScopeHandler = Mockito.mock(DataScopeHandler.class);
        processor = new DataScopeProcessor(userContextHolder, dataScopeHandler);
    }

    // === 场景1：任一角色为 ALL → 短路，不过滤 ===

    @Test
    @DisplayName("任一角色为ALL → skip=true")
    void testAnyRoleAll() {
        mockUser("zhangsan", 1L, List.of(1L, 2L));
        when(dataScopeHandler.getDataScopeTypes(List.of(1L, 2L)))
                .thenReturn(List.of(DataScopeType.ALL, DataScopeType.DEPT));

        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isTrue();
    }

    // === 场景2：本部门 + 自定义 → 合并部门 ID ===

    @Test
    @DisplayName("DEPT + CUSTOM → 合并部门ID")
    void testDeptAndCustom() {
        mockUser("zhangsan", 1L, List.of(1L, 2L));
        when(dataScopeHandler.getDataScopeTypes(List.of(1L, 2L)))
                .thenReturn(List.of(DataScopeType.DEPT, DataScopeType.CUSTOM));
        when(dataScopeHandler.getCustomDeptIds(List.of(1L, 2L)))
                .thenReturn(Set.of(10L, 20L));

        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isFalse();
        assertThat(rule.getVisibleDeptIds()).containsExactlyInAnyOrder(1L, 10L, 20L);
        assertThat(rule.isSelfOnly()).isFalse();
    }

    // === 场景3：本部门及以下 + 仅本人 → OR 条件 ===

    @Test
    @DisplayName("DEPT_AND_CHILD + SELF → 部门集 + selfOnly=true")
    void testDeptAndChildAndSelf() {
        mockUser("zhangsan", 1L, List.of(1L));
        when(dataScopeHandler.getDataScopeTypes(List.of(1L)))
                .thenReturn(List.of(DataScopeType.DEPT_AND_CHILD));
        // 触发 SELF 需要两个角色
        when(dataScopeHandler.getDataScopeTypes(List.of(1L, 2L)))
                .thenReturn(List.of(DataScopeType.DEPT_AND_CHILD, DataScopeType.SELF));
        when(dataScopeHandler.getDeptAndChildren(1L))
                .thenReturn(Set.of(1L, 2L, 3L));
        when(dataScopeHandler.getCustomDeptIds(any()))
                .thenReturn(Set.of());

        // 重新 mock 两个角色
        mockUser("zhangsan", 1L, List.of(1L, 2L));
        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isFalse();
        assertThat(rule.getVisibleDeptIds()).containsExactlyInAnyOrder(1L, 2L, 3L);
        assertThat(rule.isSelfOnly()).isTrue();
    }

    // === 场景4：仅本人 → 无部门条件 ===

    @Test
    @DisplayName("仅SELF → visibleDeptIds为空, selfOnly=true")
    void testSelfOnly() {
        mockUser("zhangsan", 1L, List.of(1L));
        when(dataScopeHandler.getDataScopeTypes(List.of(1L)))
                .thenReturn(List.of(DataScopeType.SELF));
        when(dataScopeHandler.getCustomDeptIds(any()))
                .thenReturn(Set.of());

        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isFalse();
        assertThat(rule.getVisibleDeptIds()).isEmpty();
        assertThat(rule.isSelfOnly()).isTrue();
        assertThat(rule.getUsername()).isEqualTo("zhangsan");
    }

    // === 场景5：无用户上下文（内部调用）→ 放行 ===

    @Test
    @DisplayName("无用户上下文 → skip=true")
    void testNoUserContext() {
        when(userContextHolder.get()).thenReturn(null);

        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isTrue();
    }

    // === 场景6：用户无角色 → 拒绝所有数据 ===

    @Test
    @DisplayName("用户无角色 → visibleDeptIds为空, 不skip")
    void testNoRoles() {
        mockUser("zhangsan", 1L, List.of());

        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isFalse();
        assertThat(rule.getVisibleDeptIds()).isEmpty();
        assertThat(rule.isSelfOnly()).isFalse();
        // 拦截器会将空集转为 1=0
    }

    // === 场景7：自定义但未选任何部门 → 空集 ===

    @Test
    @DisplayName("CUSTOM但未选部门 → visibleDeptIds为空")
    void testCustomEmptyDepts() {
        mockUser("zhangsan", 1L, List.of(1L));
        when(dataScopeHandler.getDataScopeTypes(List.of(1L)))
                .thenReturn(List.of(DataScopeType.CUSTOM));
        when(dataScopeHandler.getCustomDeptIds(List.of(1L)))
                .thenReturn(Set.of());

        DataScopeRule rule = processor.resolve();

        assertThat(rule.isSkip()).isFalse();
        assertThat(rule.getVisibleDeptIds()).isEmpty();
        assertThat(rule.isSelfOnly()).isFalse();
    }

    // === 辅助方法 ===

    private void mockUser(String username, Long deptId, List<Long> roleIds) {
        UserContext ctx = Mockito.mock(UserContext.class);
        when(ctx.getUsername()).thenReturn(username);
        when(ctx.getUserId()).thenReturn(1L);
        when(ctx.getDeptId()).thenReturn(deptId);
        when(ctx.getRoleIds()).thenReturn(roleIds);
        when(userContextHolder.get()).thenReturn(ctx);
    }
}
```

## 4. SQL 回归测试

### 4.1 递归 CTE 验证

```sql
-- 1. 准备部门数据
INSERT INTO sys_dept (dept_id, name, parent_id, sort_order, del_flag) VALUES
  (1, '总公司', 0, 0, '0'),
  (2, '技术部', 1, 0, '0'),
  (3, '市场部', 1, 1, '0'),
  (4, '前端组', 2, 0, '0'),
  (5, '后端组', 2, 1, '0'),
  (6, '运营部', 1, 2, '0');

-- 2. 验证递归 CTE：查询部门1的所有子部门（含自身）
WITH RECURSIVE dept_tree AS (
    SELECT dept_id FROM sys_dept WHERE dept_id = 1 AND del_flag = '0'
    UNION ALL
    SELECT d.dept_id
    FROM sys_dept d
    INNER JOIN dept_tree t ON d.parent_id = t.dept_id
    WHERE d.del_flag = '0'
)
SELECT dept_id FROM dept_tree ORDER BY dept_id;

-- 预期结果：1, 2, 3, 4, 5, 6

-- 3. 验证递归 CTE：查询部门2的子部门
WITH RECURSIVE dept_tree AS (
    SELECT dept_id FROM sys_dept WHERE dept_id = 2 AND del_flag = '0'
    UNION ALL
    SELECT d.dept_id FROM sys_dept d
    INNER JOIN dept_tree t ON d.parent_id = t.dept_id
    WHERE d.del_flag = '0'
)
SELECT dept_id FROM dept_tree ORDER BY dept_id;

-- 预期结果：2, 4, 5

-- 4. 验证空结果：查询不存在的部门
WITH RECURSIVE dept_tree AS (
    SELECT dept_id FROM sys_dept WHERE dept_id = 999 AND del_flag = '0'
    UNION ALL
    SELECT d.dept_id FROM sys_dept d
    INNER JOIN dept_tree t ON d.parent_id = t.dept_id
    WHERE d.del_flag = '0'
)
SELECT dept_id FROM dept_tree;

-- 预期结果：空集
```

### 4.2 数据权限 SQL 注入验证

```sql
-- 模拟拦截器注入后的完整 SQL（本部门及以下，deptId=1 → 子树=[1,2,3,4,5,6]）

SELECT u.user_id, u.username, u.dept_id, u.create_by
FROM sys_user u
LEFT JOIN sys_user_dept ud ON u.user_id = ud.user_id
WHERE u.del_flag = '0'
  AND ud.dept_id IN (1, 2, 3, 4, 5, 6)   -- ← 数据权限注入
GROUP BY u.user_id
ORDER BY u.create_time DESC;

-- 验证点：
-- 1. 有部门的用户只返回属于这6个部门的
-- 2. 无部门的用户（ud.dept_id 为 NULL）不返回（因为 IN 条件过滤了 NULL）
-- 3. 无重复行（GROUP BY 生效）
```

> **注意**：无部门的用户在"本部门"范围下不可见是预期行为。如果需要"无部门用户"也可见，需调整 SQL 为 `AND (ud.dept_id IN (...) OR ud.dept_id IS NULL)`，但这可能导致越权，不建议。

## 5. 边界场景测试矩阵

| # | 场景 | 预期 | 验证方法 |
|---|------|------|----------|
| 1 | 管理员角色(data_scope=1)查询 | 全部数据 | SQL 无 IN 条件 |
| 2 | 普通角色(data_scope=3)查询 | 仅本部门 | SQL 有 `dept_id = userDeptId` |
| 3 | 普通角色(data_scope=4)查询 | 本部门+子部门 | SQL 有 `dept_id IN (子树)` |
| 4 | 自定义角色(data_scope=2)无选部门 | 空结果 | SQL 有 `1=0` |
| 5 | 仅本人角色(data_scope=5)查询 | 仅自己创建 | SQL 有 `create_by = username` |
| 6 | 多角色(ALL + DEPT)查询 | 全部数据 | 短路，无 IN 条件 |
| 7 | 多角色(DEPT + SELF)查询 | 本部门 ∪ 自己 | SQL 有 `dept_id IN (...) OR create_by = ...` |
| 8 | 无用户上下文(Feign调用) | 全部数据 | 拦截器放行 |
| 9 | DataScopeContext.disable() | 全部数据 | 拦截器放行 |
| 10 | @DataScope(disabled=true) | 全部数据 | 拦截器放行 |
| 11 | 无角色用户查询 | 空结果 | SQL 有 `1=0` |
| 12 | 用户主部门为null | 仅靠其他范围 | DEPT/DEPT_AND_CHILD 跳过 |
| 13 | 部门树有环(parent_id循环引用) | CTE 不无限递归 | PG 报错或达到递归上限 |
| 14 | IN 列表超过 1000 个部门 | 性能不显著退化 | PG 支持 IN 1000+ 值 |
| 15 | SQL 中无 deptAlias 别名 | 报错或放行 | JSqlParser 解析失败时降级放行 |

### 5.1 部门循环引用防护

PostgreSQL 递归 CTE 默认有递归深度限制（`max_recursive_depth`），但 `parent_id` 循环引用会导致无限递归。

**防护措施**：

```sql
-- 方案1：SQL 中加深度限制
WITH RECURSIVE dept_tree AS (
    SELECT dept_id, parent_id, 0 AS depth
    FROM sys_dept WHERE dept_id = #{deptId} AND del_flag = '0'
    UNION ALL
    SELECT d.dept_id, d.parent_id, dt.depth + 1
    FROM sys_dept d
    INNER JOIN dept_tree dt ON d.parent_id = dt.dept_id
    WHERE d.del_flag = '0' AND dt.depth < 20  -- 深度限制
)
SELECT dept_id FROM dept_tree;

-- 方案2：应用层校验部门 parent_id 不形成环（SysDeptService.saveDept 中校验）
```
