**交互规则**：
- 数据范围下拉选择"自定义数据"时，部门树区域显示；选择其他范围时，部门树区域隐藏。
- 部门树复用已有的 `deptTree` 接口（`GET /admin/dept/tree`）。
- 打开弹窗时自动加载角色当前数据范围配置 + 已选中的部门列表。

## 2. API 对接

### 2.1 新增 API（`web/src/api/admin/role.ts`）

```typescript
/**
 * 数据范围配置接口
 */
export interface RoleDataScope {
    roleId: string;
    dataScope: string;       // '1'=全部, '2'=自定义, '3'=本部门, '4'=本部门及以下, '5'=本人
    deptIds?: string[];      // 仅 dataScope='2' 时有值
}

/**
 * 查询角色数据范围配置
 * @param roleId 角色ID
 */
export const getRoleDataScope = (roleId: string) => {
    return request({
        url: '/admin/role/data-scope/' + roleId,
        method: 'get',
    });
};

/**
 * 保存角色数据范围配置
 * @param data 数据范围配置
 */
export const saveRoleDataScope = (data: RoleDataScope) => {
    return request({
        url: '/admin/role/data-scope',
        method: 'put',
        data,
    });
};
```

## 3. 组件实现

### 3.1 新增 `data-scope.vue` 组件

文件：`web/src/views/admin/system/role/data-scope.vue`

```vue
<template>
    <div class="system-role-data-scope-container">
        <el-dialog
            v-model="visible"
            :close-on-click-modal="false"
            :title="$t('sysrole.dataScopeTitle')"
            draggable
            class="!w-[90vw] sm:!w-[480px] md:!w-[600px] !max-w-[700px]"
        >
            <el-form :model="form" label-width="90px" v-loading="loading">
                <!-- 数据范围下拉 -->
                <el-form-item :label="$t('sysrole.dataScope')">
                    <el-select v-model="form.dataScope" @change="onScopeChange" style="width: 100%">
                        <el-option
                            v-for="item in scopeOptions"
                            :key="item.value"
                            :label="$t(item.label)"
                            :value="item.value"
                        />
                    </el-select>
                </el-form-item>

                <!-- 自定义部门树（仅 dataScope='2' 时显示） -->
                <el-form-item
                    v-if="form.dataScope === '2'"
                    :label="$t('sysrole.customDept')"
                >
                    <div class="w-full">
                        <!-- 工具栏 -->
                        <div class="mb8 flex gap-3">
                            <el-checkbox
                                :label="$t('common.selectAll')"
                                @change="handleSelectAll"
                                size="small"
                            />
                            <el-checkbox
                                :label="$t('common.expand')"
                                @change="handleExpand"
                                size="small"
                            />
                        </div>
                        <!-- 部门树 -->
                        <div class="max-h-[40vh] overflow-y-auto border rounded">
                            <el-tree
                                ref="deptTreeRef"
                                :data="treeData"
                                :default-checked-keys="checkedDeptIds"
                                :props="{ label: 'name', children: 'children' }"
                                node-key="id"
                                show-checkbox
                                check-strictly
                                highlight-current
                            />
                        </div>
                    </div>
                </el-form-item>
            </el-form>

            <template #footer>
                <div class="flex justify-end gap-3">
                    <el-button @click="visible = false">
                        {{ $t('common.cancelButtonText') }}
                    </el-button>
                    <el-button type="primary" :loading="submitLoading" @click="onSubmit">
                        {{ $t('common.confirmButtonText') }}
                    </el-button>
                </div>
            </template>
        </el-dialog>
    </div>
</template>

<script setup lang="ts" name="role-data-scope">
import { getRoleDataScope, saveRoleDataScope, type RoleDataScope } from '/@/api/admin/role';
import { deptTree } from '/@/api/admin/dept';
import { useMessage } from '/@/hooks/message';
import { useI18n } from 'vue-i18n';
import type { CheckboxValueType } from 'element-plus';

const { t } = useI18n();

/** 组件引用 */
const deptTreeRef = ref();

/** 加载状态 */
const loading = ref(false);
const submitLoading = ref(false);

/** 弹窗可见性 */
const visible = ref(false);

/** 表单数据 */
const form = reactive<RoleDataScope>({
    roleId: '',
    dataScope: '1',
    deptIds: [],
});

/** 已选中的部门 ID（初始化时使用） */
const checkedDeptIds = ref<string[]>([]);

/** 部门树数据 */
const treeData = ref<any[]>([]);

/** 数据范围选项 */
const scopeOptions = [
    { value: '1', label: 'sysrole.scopeAll' },
    { value: '2', label: 'sysrole.scopeCustom' },
    { value: '3', label: 'sysrole.scopeDept' },
    { value: '4', label: 'sysrole.scopeDeptAndChild' },
    { value: '5', label: 'sysrole.scopeSelf' },
];

/**
 * 打开数据权限配置弹窗
 * @param row 角色行数据
 */
const openDialog = async (row: any) => {
    visible.value = true;
    form.roleId = row.roleId;
    form.dataScope = '1';
    form.deptIds = [];
    checkedDeptIds.value = [];
    treeData.value = [];

    loading.value = true;
    try {
        // 并行加载：角色已有配置 + 部门树
        const [scopeRes, treeRes] = await Promise.all([
            getRoleDataScope(row.roleId),
            deptTree(),
        ]);

        // 填充已有配置
        if (scopeRes.data) {
            form.dataScope = scopeRes.data.dataScope || '1';
            form.deptIds = scopeRes.data.deptIds || [];
            checkedDeptIds.value = (scopeRes.data.deptIds || []).map(String);
        }

        // 填充部门树
        treeData.value = treeRes.data || [];
    } catch (err: any) {
        useMessage().error(err.msg || t('sysrole.fetchDataScopeError'));
    } finally {
        loading.value = false;
    }
};

/**
 * 数据范围切换事件
 * 切换为非"自定义"时清空已选部门
 */
const onScopeChange = (val: string) => {
    if (val !== '2') {
        form.deptIds = [];
        deptTreeRef.value?.setCheckedKeys([]);
    }
};

/**
 * 全选/取消全选
 */
const handleSelectAll = (check: CheckboxValueType) => {
    const keys = check ? treeData.value.map((item: any) => item.id) : [];
    deptTreeRef.value?.setCheckedKeys(keys);
};

/**
 * 展开/折叠所有节点
 */
const handleExpand = (check: CheckboxValueType) => {
    const expand = (nodes: any[]) => {
        nodes.forEach((node: any) => {
            const treeNode = deptTreeRef.value?.store?.nodesMap?.[node.id];
            if (treeNode) {
                treeNode.expanded = check;
            }
            if (node.children) {
                expand(node.children);
            }
        });
    };
    expand(treeData.value);
};

/**
 * 提交数据范围配置
 */
const onSubmit = async () => {
    submitLoading.value = true;
    try {
        // 收集选中的部门 ID（仅自定义范围时）
        const selectedDeptIds =
            form.dataScope === '2'
                ? deptTreeRef.value?.getCheckedKeys() || []
                : [];

        await saveRoleDataScope({
            roleId: form.roleId,
            dataScope: form.dataScope,
            deptIds: selectedDeptIds.map(String),
        });

        useMessage().success(t('common.editSuccessText'));
        visible.value = false;
    } catch (err: any) {
        useMessage().error(err.msg || t('sysrole.saveDataScopeError'));
    } finally {
        submitLoading.value = false;
    }
};

// 暴露方法
defineExpose({
    openDialog,
});
</script>
```
